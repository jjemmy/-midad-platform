        // Copy Prompt to Clipboard
        function copyPromptText(key) {
            const promptBox = document.getElementById(`prompt-box-${key}`);
            if (!promptBox) return;

            const textToCopy = promptBox.innerText;

            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(textToCopy).then(() => showToast('تم نسخ الأمر الموجه بنجاح!'));
            } else {
                const textarea = document.createElement('textarea');
                textarea.value = textToCopy;
                document.body.appendChild(textarea);
                textarea.select();
                document.execCommand('copy');
                document.body.removeChild(textarea);
                showToast('تم نسخ الأمر الموجه بنجاح!');
            }
        }

        // Show Toast Notification
        function showToast(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-message').textContent = message;
            toast.classList.remove('translate-y-20', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 3000);
        }

        // Save & Load Plan Logic (Local Storage)
        function savePlanLocally() {
            const planData = {
                author: document.getElementById('form-author').value,
                title: document.getElementById('form-title').value,
                press: document.getElementById('form-press').value,
                posts: document.getElementById('form-posts').value,
                infographic: document.getElementById('form-infographic').value,
                script: document.getElementById('form-script').value,
                deck: document.getElementById('form-deck').value,
                date: new Date().toLocaleDateString('ar-SA')
            };

            localStorage.setItem('midad_saved_plan', JSON.stringify(planData));
            showToast('تم حفظ مسودة الخطة بنجاح في متصفحك!');
        }

        function loadSavedPlan() {
            const saved = localStorage.getItem('midad_saved_plan');
            if (saved) {
                try {
                    const data = JSON.parse(saved);
                    document.getElementById('form-author').value = data.author || '';
                    document.getElementById('form-title').value = data.title || '';
                    document.getElementById('form-press').value = data.press || '';
                    document.getElementById('form-posts').value = data.posts || '';
                    document.getElementById('form-infographic').value = data.infographic || '';
                    document.getElementById('form-script').value = data.script || '';
                    document.getElementById('form-deck').value = data.deck || '';
                } catch(e) {}
            }
        }

        function clearPlanForm() {
            if (confirm('هل أنت تأكد من إعادة تعيين النموذج؟ سيتم مسح البيانات المدخلة.')) {
                document.getElementById('interactive-plan-form').reset();
                localStorage.removeItem('midad_saved_plan');
                showToast('تم إعادة تعيين النموذج');
            }
        }
